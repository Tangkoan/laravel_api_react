ជំហានទី ១៖ បង្កើត Middleware ថ្មីមួយ
    > php artisan make:middleware CheckPermission

ជំហានទី ២៖ សរសេរកូដសម្រាប់ឆែកសិទ្ធិក្នុង Middleware
    <?php
    namespace App\Http\Middleware;
        use Closure;
        use Illuminate\Http\Request;
        use Illuminate\Support\Facades\DB;
        use Symfony\Component\HttpFoundation\Response;
        use Tymon\JWTAuth\Facades\JWTAuth;

        class CheckPermission
        {
            /**
            * @param  \Closure(\Illuminate\Http\Request): (\Symfony\Component\HttpFoundation\Response)  $next
            */
            public function handle(Request $request, Closure $next, $permissionName): Response
            {
                // ១. យក User ដែលកំពុង Login តាមរយៈ JWT Token
                try {
                    $user = JWTAuth::parseToken()->authenticate();
                } catch (\Exception $e) {
                    return response()->json(['error' => 'Unauthorized or Token Expired'], 401);
                }

                if (!$user) {
                    return response()->json(['error' => 'User not found'], 404);
                }

                // ករណីលើកលែង៖ ប្រសិនបើជា Super Admin (ឧទាហរណ៍ role_id = 1) អនុញ្ញាតឱ្យឆ្លងកាត់ដោយស្វ័យប្រវត្តិ
                // $isAdmin = DB::table('user_roles')->where('user_id', $user->id)->where('role_id', 1)->exists();
                // if ($isAdmin) return $next($request);

                // ២. ប្រើប្រាស់ SQL ដដែល ដើម្បីឆែកមើលថា តើ User នេះមានសិទ្ធិដែលយើងកំពុងទាមទារឬអត់?
                // យើងទាមទារសិទ្ធិអ្វីមួយ (ឧ. 'Product.Add') តាមរយៈអថេរ $permissionName
                $hasPermission = DB::selectOne('
                    SELECT p.id
                    FROM permissions p
                    INNER JOIN user_permission_roles pr ON p.id = pr.permission_id
                    INNER JOIN roles r ON pr.role_id = r.id
                    INNER JOIN user_roles ur ON r.id = ur.role_id
                    WHERE ur.user_id = ? AND p.name = ?
                    LIMIT 1
                ', [$user->id, $permissionName]);

                // ៣. បើរកមិនឃើញសិទ្ធិនោះទេ ទាត់គាត់ចេញ! (403 Forbidden)
                if (!$hasPermission) {
                    return response()->json([
                        'status' => 'error',
                        'message' => "Access Denied! You don't have permission: [{$permissionName}]"
                    ], 403);
                }

                // ៤. បើមានសិទ្ធិ អនុញ្ញាតឱ្យគាត់ទៅកាន់ Controller បន្តទៀត
                return $next($request);
            }
        }




ការពន្យល់កូដ Middleware:
1. កូដនេះនឹងចាប់យក JWT Token ពី Request ដើម្បីស្គាល់ថាអ្នកណាជាអ្នកបញ្ជូនមក។
2. កូដនេះនឹងចាប់យក JWT Token ពី Request ដើម្បីស្គាល់ថាអ្នកណាជាអ្នកបញ្ជូនមក។
3. ប្រើ selectOne ព្រោះយើងគ្រាន់តែចង់ដឹងថា "មាន ឬ អត់" មិនបាច់ទាញយកទាំងអស់ទេ។
4. ប្រើ selectOne ព្រោះយើងគ្រាន់តែចង់ដឹងថា "មាន ឬ អត់" មិនបាច់ទាញយកទាំងអស់ទេ។



ជំហានទី ៣៖ ចុះឈ្មោះ (Register) Middleware (សម្រាប់ Laravel 11+)
    
    => ដោយសារអ្នកកំពុងប្រើ Laravel ជំនាន់ចុងក្រោយ ការចុះឈ្មោះ Middleware ត្រូវធ្វើនៅក្នុង bootstrap/app.php។ សូមបើកវា
        ហើយបន្ថែមបន្ទាត់នេះ៖

    // នៅក្នុង bootstrap/app.php
        ->withMiddleware(function (Middleware $middleware) {
            // ចុះឈ្មោះឈ្មោះកាត់ឱ្យវា ដើម្បីងាយស្រួលហៅប្រើ
            $middleware->alias([
                'permission' => \App\Http\Middleware\CheckPermission::class,
            ]);
        })



ជំហានទី ៤៖ យក Middleware ទៅការពារ API Routes របស់អ្នក!

// នៅក្នុង routes/api.php
// ឧបមាថានេះគឺជា Group ដែលតម្រូវឱ្យ Login (មាន Token)
    Route::middleware(['auth:api'])->group(function () {
        
        // ---------------- PRODUCT ROUTES ----------------
        
        // អ្នកដែលមានសិទ្ធិ Product.View ទើបអាចទាញទិន្នន័យបាន
        Route::get('/product', [ProductController::class, 'index'])
            ->middleware('permission:Product.View');
            
        // អ្នកដែលមានសិទ្ធិ Product.Add ទើបអាចបង្កើតថ្មីបាន
        Route::post('/product', [ProductController::class, 'store'])
            ->middleware('permission:Product.Add');
            
        // អ្នកដែលមានសិទ្ធិ Product.Edit ទើបអាចកែប្រែបាន (សន្មត់ថាអ្នកប្រើ POST សម្រាប់ការកែប្រែ)
        Route::post('/product/{id}', [ProductController::class, 'update'])
            ->middleware('permission:Product.Edit');
            
        // អ្នកដែលមានសិទ្ធិ Product.Delete ទើបអាចលុបបាន
        Route::delete('/product/{id}', [ProductController::class, 'destroy'])
            ->middleware('permission:Product.Delete');
            
        // ---------------- ដាក់សម្រាប់ Route ផ្សេងៗទៀតដូចគ្នា ----------------
    });
