=> Setup

command   
        > composer require tymon/jwt-auth
        > php artisan vendor:publish --provider="Tymon\JWTAuth\Providers\LaravelServiceProvider"

        // Run command ខាងក្រោមគឺដើម្បីធ្វើការ generate key នៅ env
        > php artisan jwt:secret

* set up Authentication with JWT
    - update the config/auth.php file to use jwt as the default guard for API authentication .
        'guards' => [
            'api' => [
                'driver' => 'jwt', // កំណត់ឱ្យប្រើ jwt driver
                'provider' => 'users',
            ],
        ],
        - This tells laravel to use jwt for authentication when the api guard is used.

* update the model user
    use Tymon\JWTAuth\Contracts\JWTSubject;
    class User extends Authenticatable implements JWTSubject
    {
        // ... កូដផ្សេងៗដែលមានស្រាប់ ...

        public function getJWTIdentifier() {
            return $this->getKey();
        }

        public function getJWTCustomClaims() {
            return [];
        }
    }

* បង្កើត AuthController
    > php artisan make:controller AuthController
    ក្នុង AuthController.php អាចប្រើ auth('api')->attempt($credentials) ដើម្បីផ្ទៀងផ្ទាត់ Password និងបង្កើត Token។

* API Routes
    use App\Http\Controllers\AuthController;
    Route::group(['middleware' => 'api', 'prefix' => 'auth'], function () {
        Route::post('login', [AuthController.class, 'login']);
        Route::post('logout', [AuthController.class, 'logout']);
        Route::get('me', [AuthController.class, 'me']);
    });



* * ការកំណត់ middleware route
// ប្រើ middleware('auth:api') ផ្ទាល់តែម្តង
Route::middleware(['auth:api'])->group(function() {

    Route::controller(CategoryController::class)->group(function () {
        // ដកចេញវិញសម្រាប់តែ index និង show ដើម្បីឱ្យគេមើលបានសេរី
        Route::get('categories', 'index')->withoutMiddleware(['auth:api']);
        Route::get('categories/{id}', 'show')->withoutMiddleware(['auth:api']);
        // Route ខាងក្រោមនេះនឹងជាប់ Middleware ទាំងអស់
        Route::post('categories', 'store');
        Route::post('categories/{id}', 'update');
        Route::delete('categories/{id}', 'destroy');
        Route::put('categories/{id}/status', 'updateStatus'); 
    });
    
});


សេចក្តីសង្ខេបនៃលំហូរការងារ (Workflow)
    1. Client (React): បញ្ជូន Username/Password ទៅ API។
    2. Server (Laravel): ផ្ទៀងផ្ទាត់រួចបង្កើត Token បោះទៅឱ្យ Client វិញ។
    3. Client: រក្សាទុក Token ក្នុង LocalStorage ឬ Cookies។
    4. Request បន្ទាប់: Client ត្រូវផ្ញើ Token មកជាមួយក្នុង Header (Authorization: Bearer <token>) ដើម្បីចូលប្រើ API ដែលជាប់ Middleware។
